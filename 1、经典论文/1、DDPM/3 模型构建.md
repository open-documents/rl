$q(x_{t-1}|x_t,x_0)$

$=\frac{q(x_{t-1},x_t|x_0)}{q(x_t|x_0)}$

$=\frac  {q(x_{t-1}|x_0)  q(x_t|x_{t-1}, x_0)  }    {q(x_t|x_0)}$

$=\frac  {\textcolor{cyan}{q(x_{t-1}|x_0)}   \textcolor{orange}{q(x_t|x_{t-1})}  }    {\textcolor{green}{q(x_t|x_0)}}$          

其中：

$\textcolor{cyan}{q(x_{t-1}|x_0)}=\frac{1} {\sqrt{2Π(1-\overline{α}_{t-1})}}exp(-\frac{|x_{t-1}-\sqrt{\overline{α}_{t-1}}x_0|^{2}}{2(1-\overline{α}_{t-1})})$           $\sim \mathcal N(\sqrt{\overline{α}_{t-1}}x_0,(1-\overline{α}_{t-1})\textbf{I})$ 

$\textcolor{orange}{q(x_t|x_{t-1})} = \frac{1}{\sqrt{2Πβ_{t}}}exp(-\frac{|x_t-\sqrt{α_t}x_{t-1}|^{2}}{2β_{t}})$                      $\sim \mathcal N(\sqrt{1-β_t}x_{t-1},β_t\textbf{I})$ 

$\textcolor{green}{q(x_t|x_0)}=\frac{1} {\sqrt{2Π(1-\overline{α}_t)}}exp(-\frac{|x_t-\sqrt{\overline{α}_t}x_0|^{2}}{2(1-\overline{α}_t)})$                     $\sim \mathcal N(\sqrt{\overline{α}_t}x_0,(1-\overline{α}_t)\textbf{I})$ 

定义：$\overline{β}_t=1-\overline{α}_t$   （需要注意：

进而：

$q(x_{t-1}|x_t,x_0)=\frac{   \textcolor{orange}{\frac{1}{\sqrt{2Πβ_{t}}}exp(-\frac{|x_t-\sqrt{α_t}x_{t-1}|^{2}}{2β_{t}})} \cdot \textcolor{cyan}{\frac{1} {\sqrt{2Π(1-\overline{α}_{t-1})}}exp(-\frac{|x_{t-1}-\sqrt{\overline{α}_{t-1}}x_0|^{2}}{2(1-\overline{α}_{t-1})})}    }         {\textcolor{green}{\frac{1} {\sqrt{2Π(1-\overline{α}_t)}}exp(-\frac{|x_t-\sqrt{\overline{α}_t}x_0|^{2}}{2(1-\overline{α}_t)})}}$
